    def set_shared_history(self, node_comm, capacity):
        """
        step_control.shared_history: the ring of the five field arrays lives in ONE MPI shared-memory
        window per node (MPI-3 Win.Allocate_shared), written by the node's rank 0 and read by all;
        the other ranks skip the write in _ring_push / thin_previous and keep the indices in
        lockstep. The deposition itself is still computed on every rank (its result is identical),
        so this saves (ranks - 1) copies of the history -- 2 GB each for BC20 at 100 x 300 -- not
        time. The window is allocated once at `capacity` slices (CSR2D passes its worst-case
        retained-snapshot count); it cannot grow, so a run that needs more raises. Live slices of
        an existing private ring are copied in. Bit-identical: the same arrays, in a different
        allocation.
        """
        from mpi4py import MPI
        nx, nz = self.xbins, self.zbins
        cap = int(capacity)
        n_live = 0 if self._ring is None else self._ring_head - self._ring_tail
        if cap < n_live:
            raise ValueError(f'shared history capacity {cap} < {n_live} live slices')
        rank = node_comm.Get_rank()
        nbytes = self.NFIELD * cap * nx * nz * 8 if rank == 0 else 0
        win = MPI.Win.Allocate_shared(nbytes, 8, comm=node_comm)
        buf, itemsize = win.Shared_query(0)
        ring = np.ndarray(buffer=buf, dtype=np.float64, shape=(self.NFIELD, cap, nx, nz))
        if rank == 0 and n_live:
            ring[:, :n_live] = self._ring[:, self._ring_tail:self._ring_head]
        node_comm.Barrier()
        self._ring = ring
        self._ring_tail, self._ring_head = 0, n_live
        self._shared = (win, node_comm, rank == 0)
